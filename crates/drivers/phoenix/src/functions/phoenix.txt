# Apache Phoenix: the built-in functions of the language reference
# (phoenix.apache.org/language/functions.html), arguments as its syntax
# diagrams give them, optional ones in brackets. Every one is in Phoenix 5.0
# (checked against a 5.0 Query Server), so none is gated.

# Aggregate functions
a AVG(numericTerm)
a COUNT([DISTINCT] * | term)
a APPROX_COUNT_DISTINCT(* | term)
a MAX(term)
a MIN(term)
a SUM(numericTerm)
a PERCENTILE_CONT(numeric) WITHIN GROUP (ORDER BY numericTerm [ASC | DESC])
a PERCENTILE_DISC(numeric) WITHIN GROUP (ORDER BY numericTerm [ASC | DESC])
a PERCENT_RANK(numeric) WITHIN GROUP (ORDER BY numericTerm [ASC | DESC])
a FIRST_VALUE(expression) WITHIN GROUP (ORDER BY expression [ASC | DESC])
a LAST_VALUE(expression) WITHIN GROUP (ORDER BY expression [ASC | DESC])
a FIRST_VALUES(expression, numeric) WITHIN GROUP (ORDER BY expression [ASC | DESC])
a LAST_VALUES(expression, numeric) WITHIN GROUP (ORDER BY expression [ASC | DESC])
a NTH_VALUE(expression, nthNumeric) WITHIN GROUP (ORDER BY expression [ASC | DESC])
a STDDEV_POP(numericTerm)
a STDDEV_SAMP(numericTerm)

# Numeric functions

s CEIL(numericTerm | dateTimeTerm[, scaleNumber | DAY | HOUR | MINUTE | SECOND | MILLISECOND[, multiplierInt]])
s FLOOR(numericTerm | dateTimeTerm[, scaleNumber | DAY | HOUR | MINUTE | SECOND | MILLISECOND[, multiplierInt]])
s TRUNC(numericTerm | dateTimeTerm[, scaleNumber | DAY | HOUR | MINUTE | SECOND | MILLISECOND[, multiplierInt]])
s TO_NUMBER(stringTerm | dateTimeTerm[, formatString])
s RAND([seedNumber])
